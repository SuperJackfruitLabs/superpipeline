
### Task 35: Command palette and board switcher

This lands the command palette (`SP-Dlg-Palette-*`) and the anchored board switcher popover (`SP-Dlg-BoardSwitcher-*`), both green in light and dark. The palette indexes:
- Cards: current board, in urgency order, with `#` and the last 4 characters of the id (D-6).
- Boards: switcher order, using the `BoardSummary` counts from Task 10.
- Agents: at most 4, ranked (D-7).
- Actions and Account.

The switcher groups boards into "Needs you" and "Everything else, last used first". Its "last used" labels come from `lastActivityAt` read as calendar days in the viewer's time zone. It opens anchored under the shell's trigger, with a raised copy of that trigger above the scrim, exactly as the reference draws it.

Rulings applied:
- **Agents ranking (O-11)**, checked against the reference. The coordinator's literal rule took sole owners from every board in switcher order. That puts Agent Onboarding's onboarding-olivia and artistic-lyra (sole owners of `drafted`/`portrait`) before Production Releases, which does not match the drawn rows. The rule used here, which reproduces the reference, ranks match tiers name-starts > name-contains > capability-contains > group-only. Within a tier, in this order:
  1. agents that alone own a stage on the current board, in stage order;
  2. agents that alone own a stage on a board with no cards yet, those boards in switcher order. An empty board's stages are the only place its crew shows.
  3. the rest, by name.

  For DX with q=`agent` this gives error-eric, tester-tim, devops-dave, vulnerability-victor.

/** The calendar day of `d` in `timeZone`, as a UTC midnight timestamp. */
function dayIn(d: Date, timeZone: string): number {
  const [y, m, day] = new Intl.DateTimeFormat('en-CA', { timeZone, year: 'numeric', month: '2-digit', day: '2-digit' }).format(d).split('-').map(Number);
  return Date.UTC(y!, m! - 1, day!);
}

/** "Today", "Yesterday", "3 days ago", "Last week", "2 weeks ago", "Last month", "3 months ago": calendar days in `timeZone`. */
export function relativeDay(iso: string | null, now: Date, timeZone: string): string {
  if (!iso) return '';
  const d = Math.round((dayIn(now, timeZone) - dayIn(new Date(iso), timeZone)) / 86_400_000);
  if (d <= 0) return 'Today';
  if (d === 1) return 'Yesterday';
  if (d < 7) return `${d} days ago`;
  if (d < 14) return 'Last week';
  if (d < 28) return `${Math.floor(d / 7)} weeks ago`;
  if (d < 60) return 'Last month';
  return `${Math.floor(d / 30)} months ago`;
}

/** "Needs you" (this board first, then last used) and the rest by last use; a query filters names. */
export function groupBoards(boards: BoardSummary[], currentId: string | null, q: string): SwitcherGroup[] {
  const needle = q.trim().toLowerCase();
  const shown = needle === '' ? boards : boards.filter((b) => b.name.toLowerCase().includes(needle));
  const needs = shown.filter((b) => b.needsYouCount > 0).sort((a, b) => (a.id === currentId ? -1 : b.id === currentId ? 1 : recent(a, b)));
  const rest = shown.filter((b) => b.needsYouCount === 0).sort(recent);
  const sum = needs.reduce((n, b) => n + b.needsYouCount, 0);
  const out: SwitcherGroup[] = [];
  if (needs.length) out.push({ label: 'Needs you', right: `${sum} card${sum === 1 ? '' : 's'}`, boards: needs });
  if (rest.length) out.push({ label: needle === '' ? 'Everything else, last used first' : 'Other boards', right: `${rest.length} board${rest.length === 1 ? '' : 's'}`, boards: rest });
  return out;
}

/** The one board order the switcher and the palette share (D-18). */
export const switcherOrder = (boards: BoardSummary[], currentId: string | null): BoardSummary[] => groupBoards(boards, currentId, '').flatMap((g) => g.boards);

export const boardCountText = (n: number): string => (n === 0 ? 'No cards yet' : `${n} card${n === 1 ? '' : 's'}`);
export const needsChip = (k: number): string => (k === 1 ? '1 needs you' : `${k} need you`);

/**
 * The agents a person is most likely looking for, in order: those that alone own a stage on the
 * current board (stage order), then those that alone own a stage on a board with no cards yet,
 * those boards in switcher order. Pass the boards in that order.
 */
export function soleOwnerOrder(boards: Array<{ stages: Stage[] }>, agents: AgentSummary[]): string[] {
  const out: string[] = [];
  for (const b of boards) {
    for (const s of [...b.stages].sort((x, y) => x.order - y.order)) {
      const a = soleAgent(s, agents);
      if (a && !out.includes(a.id)) out.push(a.id);
    }
  }
  return out;
}

function agentTier(item: PaletteItem, q: string): number {
  const name = item.label.toLowerCase();
  if (q === '') return 3;
  if (name.startsWith(q)) return 0;
  if (name.includes(q)) return 1;
  if (item.sub.toLowerCase().includes(q)) return 2;
  return 3;
}

export function filterPalette(items: PaletteItem[], q: string, soleOrder: string[]): { groups: Array<{ group: PaletteGroup; items: PaletteItem[] }>; flat: PaletteItem[]; total: number } {
  const needle = q.trim().toLowerCase();
  const hit = needle === '' ? items : items.filter((i) => `${i.label} ${i.group} ${i.sub}`.toLowerCase().includes(needle));
  const rank = (i: PaletteItem) => { const k = soleOrder.indexOf(i.id.slice('agent:'.length)); return k < 0 ? Number.MAX_SAFE_INTEGER : k; };
  const groups = GROUPS.map((group) => {
    let list = hit.filter((i) => i.group === group);
    if (group === 'Agents') {
      list = [...list].sort((a, b) => agentTier(a, needle) - agentTier(b, needle) || rank(a) - rank(b) || a.label.localeCompare(b.label)).slice(0, AGENT_CAP);
    }
    return { group, items: list };
  }).filter((g) => g.items.length > 0);
  const flat = groups.flatMap((g) => g.items);
  return { groups, flat, total: flat.length };
}

export function paletteCount(n: number, q: string, boardName: string | null): string {
  if (q.trim() === '') return boardName ? `Everything on ${boardName}, plus boards, agents and actions` : 'Everything in your workspace: boards, agents and actions';
  return `${n} match${n === 1 ? '' : 'es'} for “${q.trim()}”`;
}
export const groupCount = (k: number): string => (k === 1 ? '1 result' : `${k} results`);

/** The first hit in a label, for <mark>; `hit` is '' when there is none, so no empty mark renders (D-9). */
export function splitMark(label: string, q: string): { pre: string; hit: string; post: string } {
  const needle = q.trim().toLowerCase();
  const at = needle ? label.toLowerCase().indexOf(needle) : -1;
  if (at < 0) return { pre: label, hit: '', post: '' };
  return { pre: label.slice(0, at), hit: label.slice(at, at + needle.length), post: label.slice(at + needle.length) };
}
