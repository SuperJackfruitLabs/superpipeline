
### Task 37: Reconnect flow, `POST /hub/session` and the expired dialog

This task replaces "any 401 → signed out → Landing" with recovery.

When a request is refused with 401, the store first tries a silent renewal. While it runs, a toast reads "Signing you back in…" and you stay on the page. It calls `POST /hub/session`, which spends the plane refresh cookie and re-signs the session. If the renewal fails, the store mounts an alertdialog, "Sign in again to keep going". That dialog keeps the signed-in user, names the place you will come back to, and saves `returnTo` before it sends you to sign in. This turns `SP-Dlg-Reconnect-*` green in light and dark.

Rulings applied:
- **No fixture variant.** The harness loads the page signed in, then routes `**/hub/session` to `login_required` and `**/v1/**` to 401, calls `window.__spDev.refresh()` and waits for the alertdialog.
- **Display name (D-4).** A plane session's `name` becomes `claims.name ?? claims.email`. The dialog prints the workspace member name through `personName`.
- **The silent path has no reference.** It is covered by a behavioural e2e test instead.

Decision: there is no per-request replay. Once renewal succeeds, the store reloads the person and the board (`#restore`) and reconnects the socket. A second 401 during that reload ends in `expired` with `unauthorized_after_renew`.

```ts
function plane(answer: () => Record<string, unknown> | null) {
  const tokenRequests: URLSearchParams[] = [];
  const impl = (async (input: RequestInfo | URL, init?: RequestInit) => {
    const url = String(input);
    if (url === PLANE_JWKS) return new Response((await planeKeys()).jwksBody, { headers: { 'content-type': 'application/json' } });
    if (url === `${PLANE}/api/auth/oauth2/token`) {

      const body = answer();
      return body ? Response.json(body) : Response.json({ error: 'invalid_grant' }, { status: 400 });
    }
    return new Response('unexpected', { status: 599 });
  }) as unknown as typeof fetch;
  return { impl, tokenRequests };
}
function cookies(res: Response): Map<string, string> {
  const out = new Map<string, string>();
  for (const c of (res.headers as unknown as { getSetCookie(): string[] }).getSetCookie()) {
    const [pair] = c.split(';');
    const [k, ...v] = pair!.trim().split('=');
    out.set(k!, v.join('='));
  }
  return out;
}
```

```ts
import { describe, expect, it, vi } from 'vitest';
import { AuthRecovery, RETRY_MS } from './auth-recovery.svelte';
import type { RenewResult } from './api';

function make(renewals: RenewResult[], restore: () => Promise<boolean> = async () => true) {
  const renew = vi.fn(async () => renewals.shift() ?? { ok: false as const, reason: 'login_required' as const });
  const sleep = vi.fn(async () => {});
  const onExpired = vi.fn();
  return { a: new AuthRecovery({ renew, restore: vi.fn(restore), sleep, onExpired }), renew, sleep, onExpired };
}

describe('AuthRecovery', () => {
  it('concurrent 401s share one renewal', async () => {
    const { a, renew } = make([{ ok: true }]);
    const p1 = a.recover(); const p2 = a.recover();
    a.noteUnauthorized();
    expect(a.state).toBe('renewing');
    expect(await p1).toBe(true);
    expect(await p2).toBe(true);
    expect(renew).toHaveBeenCalledTimes(1);
    expect(a.state).toBe('ok');
  });
  it('a renewal that succeeds restores once and stays ok', async () => {
    const restore = vi.fn(async () => true);
    const a = new AuthRecovery({ renew: async () => ({ ok: true }), restore, sleep: async () => {} });
    expect(await a.recover()).toBe(true);
    expect(restore).toHaveBeenCalledTimes(1);
    expect([a.state, a.reason]).toEqual(['ok', null]);
  });
  it('a second 401 while restoring ends expired: refused after renew', async () => {
    let a!: AuthRecovery;
    const made = make([{ ok: true }], async () => { a.noteUnauthorized(); return true; });
    a = made.a;
    expect(await a.recover()).toBe(false);
    expect([a.state, a.reason]).toEqual(['expired', 'unauthorized_after_renew']);
    expect(made.renew).toHaveBeenCalledTimes(1);
  });
  it('login_required ends expired with the reason and the issuer', async () => {
    const { a, onExpired } = make([{ ok: false, reason: 'login_required', issuer: 'accounts.superjackfruit.com', reauthorize: '/auth/login' }]);
    expect(await a.recover()).toBe(false);
    expect([a.state, a.reason, a.issuer]).toEqual(['expired', 'login_required', 'accounts.superjackfruit.com']);
    expect(onExpired).toHaveBeenCalledTimes(1);
  });
  it('an unavailable sign-in service is retried three times, five seconds apart, then expires', async () => {
    const down = { ok: false as const, reason: 'plane_unavailable' as const, retryable: true };
    const { a, renew, sleep } = make([down, down, down, down]);
    expect(await a.recover()).toBe(false);
    expect(renew).toHaveBeenCalledTimes(4);
    expect(sleep.mock.calls).toEqual([[RETRY_MS], [RETRY_MS], [RETRY_MS]]);
    expect(RETRY_MS).toBe(5000);
    expect(a.reason).toBe('plane_unavailable');
  });
  it('once expired, further 401s do nothing until resolved', async () => {
    const { a, renew } = make([{ ok: false, reason: 'no_refresh' }]);
    await a.recover();
    a.noteUnauthorized();
    expect(renew).toHaveBeenCalledTimes(1);
    a.resolved();
    expect([a.state, a.reason, a.issuer]).toEqual(['ok', null, null]);
  });
});
```

```ts
export function displayName(claims: { name?: unknown; email?: string }): string {
  return typeof claims.name === 'string' && claims.name.trim() !== '' ? claims.name.trim() : claims.email ?? 'Member';

/** A signed session for a renewed token, or null when it is not a person who may sign in here. */
async function sessionFromClaims(env: Env, claims: OrgPlaneClaims): Promise<string | null> {
  if (!entitles(claims) || claims.principalKind !== 'human' || claims.act) return null;
  const tenantId = await ensureOrgTenant(env.DB, claims.org);
  const human = await provisionOrgHuman(env.DB, tenantId, claims);
  if (!human) return null;
  return signSession({ userId: human.userId, tenantId, name: displayName(claims), exp: Date.now() + SESSION_TTL_MS }, env.SESSION_SECRET!);
}

/**
 * POST /hub/session: renew a lapsed session without leaving the page. It lives under /hub/ because the
 * refresh cookie is Path=/hub. It spends the refresh cookie, re-signs the session and rotates the
 * cookies exactly as /hub/token does. A refused grant is NOT a sign-out, so the refresh cookie is
 * left for another tab that may already hold a newer one. GitHub sign-in has no refresh token.
 */
export async function handleSessionRenewal(
  request: Request,
  env: Env,
  fetchImpl: typeof fetch = fetch,
  sleep: (ms: number) => Promise<void> = (ms) => new Promise((r) => setTimeout(r, ms)),
): Promise<Response> {
  if (request.method !== 'POST') return Response.json({ error: 'method not allowed' }, { status: 405 });
  const mode = orgPlaneMode(env);
  if (mode.kind === 'off') {
    if (env.DEV_AUTH === 'true') {
      // Dev stub, steered by the refresh cookie: `refused` and `slow` let e2e reach both outcomes.
      const v = readCookie(request, REFRESH_COOKIE);
      if (v === 'refused') return Response.json({ ok: false, reason: 'login_required', issuer: DEV_ISSUER, reauthorize: '/auth/login' });
      if (v === 'slow') await sleep(10_000);
      return Response.json({ ok: true });
    }
    return Response.json({ ok: false, reason: 'no_refresh' });
  }
  if (mode.kind === 'invalid' || !env.SESSION_SECRET) return unavailable();
  const { cfg } = mode;
  const issuer = new URL(cfg.issuer).host;
  const refused = () => Response.json({ ok: false, reason: 'login_required', issuer, reauthorize: '/auth/login' });

  const refresh = readCookie(request, REFRESH_COOKIE);
  if (!refresh) return Response.json({ ok: false, reason: 'no_refresh' });
  const grant = await tokenGrant(cfg, { grant_type: 'refresh_token', refresh_token: refresh }, fetchImpl);
  if (!grant.ok) return grant.refused ? refused() : unavailable();
  const tokens = grant.tokens;
  const claims = await verifyOrgPlaneToken(tokens.access_token!, cfg, cfg.audience, fetchImpl);
  const session = claims ? await sessionFromClaims(env, claims) : null;
  if (!session) return refused();

  const headers = new Headers();
  headers.append('Set-Cookie', sessionSetCookie(session, { secure: true }));
  headers.append('Set-Cookie', tokenCookie(tokens.access_token!, ttlOf(tokens)));
  let next: string | null = tokens.refresh_token ?? null;
  if (next) {
    const hub = await hubAudienceToken(cfg, env, next, fetchImpl);
    if (hub) {
      headers.append('Set-Cookie', hubAudCookie(hub.access_token!, ttlOf(hub)));
      if (hub.refresh_token) next = hub.refresh_token;
    }
  }
  headers.append('Set-Cookie', refreshCookie(next ?? '', next ? REFRESH_MAX_AGE_S : 0));
  return Response.json({ ok: true }, { headers });
}
```

```ts
// apps/web/src/lib/auth-recovery.svelte.ts
/**
 * What happens when the server says the session is gone. It renews quietly first (POST /hub/session,
 * one renewal shared by every concurrent 401), then restores the page. Only when that fails does the
 * app ask the person to sign in again. The person stays signed in on screen throughout: Landing is
 * for people who were never signed in.
 */
import { renewSession, type RenewReason, type RenewResult } from './api';

export type AuthRecoveryState = 'ok' | 'renewing' | 'expired';
export const RETRY_MS = 5000;
export const RETRIES = 3;

export interface RecoveryDeps {
  renew: () => Promise<RenewResult>;
  /** Reload the person and the board after a renewal; false when the person cannot be read back. */
  restore: () => Promise<boolean>;
  sleep: (ms: number) => Promise<void>;
  onExpired?: () => void;
}

export class AuthRecovery {
  state = $state<AuthRecoveryState>('ok');
  reason = $state<RenewReason | null>(null);
  issuer = $state<string | null>(null);
  #deps: RecoveryDeps;
  #inflight: Promise<boolean> | null = null;
  #restoring = false;
  #refusedAgain = false;

  constructor(deps: RecoveryDeps) { this.#deps = deps; }
  configure(deps: Partial<RecoveryDeps>): void { this.#deps = { ...this.#deps, ...deps }; }

  noteUnauthorized(): void {
    if (this.state === 'expired') return;
    if (this.#restoring) { this.#refusedAgain = true; return; }
    void this.recover();
  }

  recover(): Promise<boolean> {
    this.#inflight ??= this.#run().finally(() => { this.#inflight = null; });
    return this.#inflight;
  }

  /** Signed in again elsewhere (another tab): nothing to recover any more. */
  resolved(): void {
    this.state = 'ok';
    this.reason = null;
    this.issuer = null;
  }

  async #run(): Promise<boolean> {
    this.state = 'renewing';
    let r = await this.#deps.renew();
    for (let i = 0; !r.ok && r.reason === 'plane_unavailable' && i < RETRIES; i++) {
      await this.#deps.sleep(RETRY_MS);
      r = await this.#deps.renew();
    }
    if (!r.ok) return this.#expire(r.reason, r.issuer ?? null);
    this.#restoring = true;
    this.#refusedAgain = false;
    let restored = false;
    try { restored = await this.#deps.restore(); } finally { this.#restoring = false; }
    if (!restored || this.#refusedAgain) return this.#expire('unauthorized_after_renew', null);
    this.resolved();
    return true;
  }

  #expire(reason: RenewReason, issuer: string | null): false {
    this.state = 'expired';
    this.reason = reason;
    this.issuer = issuer;
    this.#deps.onExpired?.();
    return false;
  }
}

export const auth = new AuthRecovery({
  renew: renewSession,
  restore: async () => false,
  sleep: (ms) => new Promise((r) => setTimeout(r, ms)),
});
```
