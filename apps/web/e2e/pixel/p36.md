
### Task 36: Appearance dialog, with the product accent

This lands the Appearance dialog (`SP-Dlg-Appearance-*`), which turns green in light and dark at 1440×1200. The dialog has:

The dialog runs on Task 1's `appearance.accent`/`setAccent`. Every control applies at once.

```ts
/** SP-Dlg-Appearance-*: the canvas resolved the theme control at export, so dark checks "Dark" and changes its hint. */
export function appearanceDarkPatch(doc: Document): void {
  const group = doc.querySelector('[role="radiogroup"][aria-label="Light or dark"]');
  if (!group) return;
  for (const b of group.querySelectorAll<HTMLElement>('[role="radio"]')) {
    const on = b.textContent?.trim() === 'Dark';
    b.setAttribute('aria-checked', String(on));
    b.setAttribute('data-state', on ? 'checked' : 'unchecked');
  }
  const hint = group.parentElement?.querySelector('.ap-hint');
  if (hint) hint.textContent = 'Always dark: a lamp-lit room, not black.';
}
```

```ts
import { greeting, type Phase, type Vibe } from '@superjackfruit/vibekit';
import type { Accent, AccentChoice } from '$lib/accent';

export const PHASE_NAMES: Readonly<Record<Phase, string>> = { dawn: 'Dawn', morning: 'Morning', noon: 'Midday', golden: 'Golden hour', dusk: 'Dusk', night: 'Night' };

export const THEME_OPTIONS = [{ value: 'sun', label: 'Follow the sun' }, { value: 'light', label: 'Light' }, { value: 'dark', label: 'Dark' }] as const;
export const STRENGTH_OPTIONS = [{ value: 'strong', label: 'Strong' }, { value: 'subtle', label: 'Subtle' }, { value: 'off', label: 'Off' }] as const;

export function themeHint(pref: 'sun' | 'light' | 'dark', phase: Phase, resolved: 'light' | 'dark'): string {
  if (pref === 'light') return 'Always light, whatever the hour.';
  if (pref === 'dark') return 'Always dark: a lamp-lit room, not black.';
  return `Light by day, dark after dusk. It is ${PHASE_NAMES[phase].toLowerCase()}, so it is ${resolved} now.`;
}

const TIME_HINT = {
  strong: 'Strong: the full sky and greeting where the vibe shows them.',
  subtle: 'Subtle: a light tint of the sky, no sun.',
  off: 'Off: no sky, no tint. Light or dark still applies.',
} as const;
export const timeHint = (s: keyof typeof TIME_HINT): string => `${TIME_HINT[s]} Quiet starts with this off.`;

/** vibekit's phaseAt day window without a location (no exported constant; documented 06:00–18:00). */
const DAY_WINDOW = '6:00 to 18:00';
export function locMeans(on: boolean, tz: string): string {
  return on
    ? 'On: sunrise and sunset are worked out from your rough location (about 11 km). It stays on this device and is never sent anywhere.'
    : `Off: the sun follows your time zone, ${tz}, and the day runs ${DAY_WINDOW}. Turn it on for real sunrise and sunset; your browser will ask first.`;
}
export const LOC_DENIED = 'Location was not shared, so the sun follows your time zone.';

/** The sky readout's line (D-11). `hhmm` is the local clock, for Studio. */
export function readout(vibe: Vibe, phase: Phase, strength: 'strong' | 'subtle' | 'off', first: string, hhmm: string): string {
  const P = PHASE_NAMES[phase];
  const line =
    vibe === 'daylight' ? `${P} · ${greeting(phase, 'daylight').replace(/\.$/, '')}, ${first}.`
    : vibe === 'paper' ? (phase === 'noon' ? `${P} · The light is high and even.` : `${P} · ${greeting(phase, 'paper')}`)
    : vibe === 'studio' ? `${P}, ${hhmm} · the board at a glance`
    : `${P}.`;
  return (strength === 'off' ? 'Time of day is off · ' : '') + line;
}
```

```svelte
  import './ap-extra.css';

  const ZONE = Intl.DateTimeFormat().resolvedOptions().timeZone;
  let locNote = $state<string | null>(null);
  let locBusy = $state(false);

  const first = $derived(firstName(personName(app.user, app.members)));
  const counts = $derived.by(() => {
    const b = app.board;
    if (!b) return { needs: 0, working: 0, queued: 0 };
    const s = summarize(b, b.cards.filter((c) => !c.archivedAt), { agents: app.agents, members: app.members, labels: app.labelById(), nowMs: Date.now() });
    return { needs: s.filter((x) => x.urgency === 'needs').length, working: s.filter((x) => x.urgency === 'working').length, queued: s.filter((x) => x.urgency === 'queued').length };
  });
  const on = $derived(appearance.value.useLocation || locBusy);

  function close(): void { appearance.pickerOpen = false; }

  async function flipLocation(): Promise<void> {
    // Off wins: while a request is pending the switch reads on, and a second press cancels it.
    if (locBusy || appearance.value.useLocation) {
      appearance.stopUsingLocation();
      locBusy = false;
      locNote = null;
      return;
    }
    locBusy = true;
    locNote = null;
    const r = await appearance.useLocation();
    if (r === 'cancelled') return;
    locBusy = false;
    locNote = r === 'denied' ? LOC_DENIED : null;
  }
```
